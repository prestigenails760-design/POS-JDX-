async function saveAppointment() { let name = $('#aName').value.trim(); let phone = $('#aPhone').value.trim(); let bm = $('#aBirthMonth').value; let bd = $('#aBirthDay').value; let date = $('#aDate').value; let time24 = $('#aTime').value; if (!name || !phone || !bm || !bd || !date || !time24) { return toast("Please complete all required fields"); } const body = { customer_name: name, phone: phone, birthday_month: bm, birthday_day: bd, service_name: $('#aService').value, technician: $('#aTech').value, appointment_date: date, appointment_time: time24, duration: Number($('#aDuration').value), buffer: Number($('#aBuffer').value), deposit: Number($('#aDeposit').value), status: $('#aStatus').value }; try { const r = await fetch("/api/create-booking",{ method:"POST", headers:{ "Content-Type":"application/json" }, body:JSON.stringify(body) }); const data = await r.json(); if(!r.ok){ console.log(data); return toast("Booking failed"); } toast("Appointment Saved"); closeModal(); loadAppointments(); } catch(e){ console.error(e); toast("Server Error"); } }
